# MacVPN Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Natív macOS menüsor-app, amely egy helyen kezeli az OpenVPN, WireGuard, IKEv2 (strongSwan), L2TP és Cisco IPsec kapcsolatokat. Config szerkesztés, Keychain, élő/debug napló, DMG terjesztés.

**Architecture:** SwiftPM projekt két targettel: `MacVPNCore` (tesztelhető logika: modellek, validátorok, store-ok, backendek, parserek) és `MacVPN` (SwiftUI `MenuBarExtra` app). A root műveleteket egyetlen `mvpn` bash dispatcher végzi (`/usr/local/libexec/macvpn/mvpn`), amit az app `sudo -n`-nel hív egy sudoers szabály alapján. Natív VPN-ek `scutil --nc`-vel és generált `.mobileconfig`-gal.

**Tech Stack:** Swift 5 language mode (swift-tools 5.10, Xcode toolchain 6.x), SwiftUI, XCTest, Security.framework (Keychain), bash 3.2 kompatibilis root scriptek, Homebrew (`openvpn`, `wireguard-tools`, `strongswan`, `bash`), `hdiutil`.

**Spec:** `docs/superpowers/specs/2026-10-16-macvpn-design.md`

```swift
public struct Connection: Codable, Identifiable, Hashable, Sendable {
    public var id: UUID
    public var name: String
    public var kind: VPNKind
    public var createdAt: Date

    public var native: NativeSettings?

    public init(id: UUID = UUID(), name: String, kind: VPNKind, createdAt: Date = Date(),
                nativeServiceName: String? = nil, native: NativeSettings? = nil) {
        self.id = id
        self.name = name
        self.kind = kind
        self.createdAt = createdAt
        self.nativeServiceName = nativeServiceName
        self.native = native
    }

public enum ConnectionState: Equatable, Sendable {
    case disconnected
    case connecting
    case connected(ip: String?, since: Date?)
    case disconnecting
    case failed(String)

    public var isActive: Bool {
        switch self {
        case .connecting, .connected, .disconnecting: return true
        case .disconnected, .failed: return false
        }
    }

    public var isConnected: Bool {
        if case .connected = self { return true }
        return false
    }
}

/// `line == 0`: az egész fájlra vonatkozó hiba.
public struct ValidationIssue: Equatable, Sendable {
    public let line: Int
    public let message: String

    public init(line: Int, message: String) {
        self.line = line
        self.message = message
    }

    public var display: String { line == 0 ? message : "\(line). sor: \(message)" }
}

public struct Secrets: Equatable, Sendable {
    public var username: String?
    public var password: String?
    public var psk: String?

    public init(username: String? = nil, password: String? = nil, psk: String? = nil) {
        self.username = username
        self.password = password
        self.psk = psk
    }

    public var isEmpty: Bool { stdinPayload.isEmpty }

    /// A `mvpn up` stdin formátuma: `kulcs=érték` soronként.
    public var stdinPayload: String {
        var out = ""
        for (key, value) in [("username", username), ("password", password), ("psk", psk)] {
            guard let v = value?.replacingOccurrences(of: "\n", with: "")
                .replacingOccurrences(of: "\r", with: ""), !v.isEmpty else { continue }
            out += "\(key)=\(v)\n"
        }
        return out
    }
}
```

```swift
import Foundation

public protocol ConfigValidator {
    func validate(_ text: String) -> [ValidationIssue]
}

public enum Validators {
    public static func validator(for kind: VPNKind) -> ConfigValidator? {
        switch kind {
        case .openvpn: return OpenVPNValidator()
        case .wireguard: return WireGuardValidator()
        case .strongswan: return StrongSwanValidator()
        case .l2tp, .cisco: return nil
        }
    }
}

/// Soronként, CR és szélső whitespace nélkül, 1-től számozva.
func numberedLines(_ text: String) -> [(Int, String)] {
    text.components(separatedBy: "\n").enumerated().map { index, raw in
        (index + 1, raw.trimmingCharacters(in: CharacterSet.whitespaces.union(CharacterSet(charactersIn: "\r"))))
    }
}

/// A szabályok tükrözik a `Resources/root/mvpn` validate_openvpn függvényét. Változtatáskor mindkettőt frissítsd.
public struct OpenVPNValidator: ConfigValidator {
    public static let forbidden: Set<String> = [
        "up", "down", "route-up", "route-pre-down", "ipchange", "tls-verify", "auth-user-pass-verify",
        "client-connect", "client-disconnect", "learn-address", "plugin", "script-security", "config",
        "cd", "chroot", "daemon", "log", "log-append", "writepid", "management", "status", "setenv",
        "iproute", "ifconfig-noexec", "route-noexec", "user", "group",
    ]
    public static let fileReferencing: Set<String> = [
        "ca", "cert", "key", "tls-auth", "tls-crypt", "tls-crypt-v2", "pkcs12", "secret",
        "extra-certs", "dh", "crl-verify", "askpass", "auth-user-pass",
    ]

    public init() {}

    public func validate(_ text: String) -> [ValidationIssue] {
        var issues: [ValidationIssue] = []
        var inBlock: String?
        var hasRemote = false
        for (lineNo, line) in numberedLines(text) {
            if let block = inBlock {
                if line.lowercased() == "</\(block)>" { inBlock = nil }
                continue
            }
            if line.range(of: "^<[A-Za-z0-9-]+>$", options: .regularExpression) != nil {
                inBlock = String(line.dropFirst().dropLast()).lowercased()
                continue
            }
            if line.isEmpty || line.hasPrefix("#") || line.hasPrefix(";") { continue }
            let tokens = line.split(whereSeparator: { $0 == " " || $0 == "\t" }).map(String.init)
            var directive = tokens[0].lowercased()
            if directive.hasPrefix("--") { directive.removeFirst(2) }
            if Self.forbidden.contains(directive) {
                issues.append(ValidationIssue(line: lineNo, message: "Tiltott direktíva: \(directive)"))
                continue
            }
            if Self.fileReferencing.contains(directive), tokens.count >= 2, tokens[1] != "[inline]" {
                issues.append(ValidationIssue(line: lineNo, message: "Külső fájl hivatkozás, inline blokk kell: \(directive)"))
            }
            if directive == "remote" { hasRemote = true }
        }
        if !hasRemote { issues.append(ValidationIssue(line: 0, message: "Hiányzik a remote direktíva")) }
        return issues
    }
}
```

```swift
/// Tükrözi a `mvpn` validate_wireguard függvényét.
public struct WireGuardValidator: ConfigValidator {
    public static let forbiddenKeys: Set<String> = ["preup", "postup", "predown", "postdown", "saveconfig"]

    public init() {}

    public func validate(_ text: String) -> [ValidationIssue] {
        var issues: [ValidationIssue] = []
        var section = ""
        var interfaceCount = 0, peerCount = 0
        var hasPrivateKey = false, hasPeerKey = false
        for (lineNo, rawLine) in numberedLines(text) {
            let line = rawLine.split(separator: "#", maxSplits: 1, omittingEmptySubsequences: false)
                .first.map { $0.trimmingCharacters(in: .whitespaces) } ?? ""
            if line.isEmpty { continue }
            if line.hasPrefix("["), line.hasSuffix("]") {
                section = line.lowercased()
                if section == "[interface]" { interfaceCount += 1 }
                if section == "[peer]" { peerCount += 1 }
                continue
            }
            let key = line.split(separator: "=", maxSplits: 1).first
                .map { $0.trimmingCharacters(in: .whitespaces).lowercased() } ?? ""
            if Self.forbiddenKeys.contains(key) {
                issues.append(ValidationIssue(line: lineNo, message: "Tiltott kulcs: \(key)"))
            }
            if section == "[interface]", key == "privatekey" { hasPrivateKey = true }
            if section == "[peer]", key == "publickey" { hasPeerKey = true }
        }
        if interfaceCount != 1 { issues.append(ValidationIssue(line: 0, message: "Pontosan egy [Interface] szakasz kell")) }
        if !hasPrivateKey && interfaceCount == 1 { issues.append(ValidationIssue(line: 0, message: "Hiányzik a PrivateKey")) }
        if peerCount < 1 || !hasPeerKey { issues.append(ValidationIssue(line: 0, message: "Legalább egy [Peer] kell PublicKey-jel")) }
        return issues
    }
}

/// Tükrözi a `mvpn` validate_strongswan és swan_names függvényét.
public struct StrongSwanValidator: ConfigValidator {
    public init() {}

    public func validate(_ text: String) -> [ValidationIssue] {
        var issues: [ValidationIssue] = []
        var hasConnections = false, hasChildren = false
        for (lineNo, rawLine) in numberedLines(text) {
            let line = Self.stripComment(rawLine)
            if line.isEmpty { continue }
            if line.range(of: "^include[ \\t]", options: .regularExpression) != nil {
                issues.append(ValidationIssue(line: lineNo, message: "Tiltott: include"))
                continue
            }
            if line.range(of: "^connections[ \\t]*\\{", options: .regularExpression) != nil { hasConnections = true }
            if line.range(of: "^children[ \\t]*\\{", options: .regularExpression) != nil { hasChildren = true }
            for statement in line.components(separatedBy: CharacterSet(charactersIn: "{}")) {
                guard let eq = statement.firstIndex(of: "=") else { continue }
                let key = statement[..<eq].trimmingCharacters(in: .whitespaces).lowercased()
                if key == "updown" || key == "file" {
                    issues.append(ValidationIssue(line: lineNo, message: "Tiltott kulcs: \(key)"))
                }
            }
        }
        if !hasConnections { issues.append(ValidationIssue(line: 0, message: "Hiányzik a connections blokk")) }
        if !hasChildren { issues.append(ValidationIssue(line: 0, message: "Hiányzik a children blokk")) }
        return issues

    /// Az első connection és az első child neve (a `swanctl --initiate --child` paraméteréhez).
    public static func names(in text: String) -> (connection: String, child: String)? {
        var inConnections = false, inChildren = false
        var connection: String?, child: String?
        let header = try! NSRegularExpression(pattern: "^([A-Za-z0-9_.-]+)[ \\t]*\\{")
        for (_, rawLine) in numberedLines(text) {
            let line = stripComment(rawLine)
            if line.range(of: "^connections[ \\t]*\\{", options: .regularExpression) != nil { inConnections = true; continue }
            if line.range(of: "^children[ \\t]*\\{", options: .regularExpression) != nil { inChildren = true; continue }
            let ns = line as NSString
            guard let m = header.firstMatch(in: line, range: NSRange(location: 0, length: ns.length)) else { continue }
            let name = ns.substring(with: m.range(at: 1))
            if inConnections, connection == nil { connection = name }
            if inChildren, child == nil { child = name }
        }
        guard let c = connection, let ch = child else { return nil }
        return (c, ch)
    }

    static func stripComment(_ line: String) -> String {
        (line.split(separator: "#", maxSplits: 1, omittingEmptySubsequences: false).first.map(String.init) ?? "")
            .trimmingCharacters(in: .whitespaces)
    }
```

```swift
import Foundation

public enum ConfigImportError: LocalizedError, Equatable {
    case missingFile(String)
    case unsupported(String)
    case unknownType

    public var errorDescription: String? {
        switch self {
        case .missingFile(let f): return "A hivatkozott fájl nem található: \(f)"
        case .unsupported(let d): return "Nem támogatott direktíva importnál: \(d)"
        case .unknownType: return "Nem ismerhető fel a config típusa"
        }
    }
}

public enum ConfigImporter {
    public static func detectKind(fileName: String, content: String) -> VPNKind? {
        let ext = (fileName as NSString).pathExtension.lowercased()
        if ext == "ovpn" { return .openvpn }
        let lower = content.lowercased()
        if lower.contains("[interface]"), lower.contains("privatekey") { return .wireguard }
        if content.range(of: "(?m)^\\s*connections\\s*\\{", options: .regularExpression) != nil { return .strongswan }
        if content.range(of: "(?m)^\\s*remote\\s+\\S+", options: .regularExpression) != nil,
           content.range(of: "(?m)^\\s*(client|dev\\s+tun)", options: .regularExpression) != nil { return .openvpn }
        return nil
    }

    static let inlineable: Set<String> = ["ca", "cert", "key", "tls-auth", "tls-crypt", "tls-crypt-v2", "extra-certs", "dh"]

    /// A külső fájlra hivatkozó OpenVPN direktívákat inline blokká alakítja.
    public static func inlineOpenVPNFiles(_ content: String, baseDirectory: URL) throws -> String {
        var out: [String] = []
        for raw in content.components(separatedBy: "\n") {
            let line = raw.trimmingCharacters(in: CharacterSet.whitespaces.union(CharacterSet(charactersIn: "\r")))
            let tokens = line.split(whereSeparator: { $0 == " " || $0 == "\t" }).map(String.init)
            guard let first = tokens.first?.lowercased(), tokens.count >= 2, tokens[1] != "[inline]" else {
                out.append(raw)
                continue
            }
            if first == "auth-user-pass" {
                out.append("auth-user-pass")
                continue
            }
            if first == "pkcs12" { throw ConfigImportError.unsupported("pkcs12") }
            guard inlineable.contains(first) else {
                out.append(raw)
                continue
            }
            let path = tokens[1].trimmingCharacters(in: CharacterSet(charactersIn: "\"'"))
            let url = path.hasPrefix("/") ? URL(fileURLWithPath: path) : baseDirectory.appendingPathComponent(path)
            guard let data = try? String(contentsOf: url, encoding: .utf8) else {
                throw ConfigImportError.missingFile(path)
            }
            out.append("<\(first)>")
            out.append(data.hasSuffix("\n") ? String(data.dropLast()) : data)
            out.append("</\(first)>")
            if first == "tls-auth", tokens.count >= 3 { out.append("key-direction \(tokens[2])") }
        }
        return out.joined(separator: "\n")
    }

    public static func template(for kind: VPNKind) -> String {
        switch kind {
        case .openvpn:
            return """
            client
            dev tun
            proto udp
            remote vpn.example.com 1194
            nobind
            persist-key
            persist-tun
            remote-cert-tls server
            auth-user-pass
            <ca>
            </ca>

            """
        case .wireguard:
            return """
            [Interface]
            PrivateKey =
            Address = 10.0.0.2/32
            DNS = 1.1.1.1

            [Peer]
            PublicKey =
            AllowedIPs = 0.0.0.0/0
            Endpoint = vpn.example.com:51820

            """
        case .strongswan:
            return """
            connections {
              myvpn {
                remote_addrs = vpn.example.com
                vips = 0.0.0.0
                local {
                  auth = eap-mschapv2
                  eap_id = felhasznalo

                remote {

                  id = vpn.example.com
                }
                children {
                  myvpn {
                    remote_ts = 0.0.0.0/0
                  }
                }
              }
            }

            """
        case .l2tp, .cisco:
            return ""
        }
    }
```

```swift
import Foundation

/// `<base>/<UUID>/meta.json`, `config`, `profile.mobileconfig`.
/// A `mvpn` root script ugyanebből az útvonalból olvas: ~/Library/Application Support/MacVPN/connections/<UUID>/config
public final class ConnectionStore {
    public let baseURL: URL

    public static var defaultBaseURL: URL {
        FileManager.default.homeDirectoryForCurrentUser
            .appendingPathComponent("Library/Application Support/MacVPN/connections", isDirectory: true)
    }

    private let fm = FileManager.default
    private let encoder: JSONEncoder = {
        let e = JSONEncoder()
        e.dateEncodingStrategy = .iso8601
        e.outputFormatting = [.prettyPrinted, .sortedKeys]
        return e
    }()
    private let decoder: JSONDecoder = {
        let d = JSONDecoder()
        d.dateDecodingStrategy = .iso8601
        return d
    }()

    public init(baseURL: URL = ConnectionStore.defaultBaseURL) {
        self.baseURL = baseURL
    }

    public func directory(for id: UUID) -> URL { baseURL.appendingPathComponent(id.uuidString, isDirectory: true) }
    public func configURL(for id: UUID) -> URL { directory(for: id).appendingPathComponent("config") }
    public func profileURL(for id: UUID) -> URL { directory(for: id).appendingPathComponent("profile.mobileconfig") }
    func metaURL(for id: UUID) -> URL { directory(for: id).appendingPathComponent("meta.json") }

    public func list() throws -> [Connection] {
        guard fm.fileExists(atPath: baseURL.path) else { return [] }
        let dirs = try fm.contentsOfDirectory(at: baseURL, includingPropertiesForKeys: nil)
        return dirs.compactMap { dir in
            guard let data = try? Data(contentsOf: dir.appendingPathComponent("meta.json")) else { return nil }
            return try? decoder.decode(Connection.self, from: data)
        }
        .sorted { $0.name.localizedCaseInsensitiveCompare($1.name) == .orderedAscending }
    }

    /// `config == nil`: a meglévő config fájl változatlan marad.
    public func save(_ connection: Connection, config: String?) throws {
        let dir = directory(for: connection.id)
        try fm.createDirectory(at: dir, withIntermediateDirectories: true, attributes: [.posixPermissions: 0o700])
        try encoder.encode(connection).write(to: metaURL(for: connection.id), options: .atomic)
        if let config {
            try writePrivate(Data(config.utf8), to: configURL(for: connection.id))
        }
    }

    public func writeProfile(_ data: Data, for id: UUID) throws {
        try writePrivate(data, to: profileURL(for: id))
    }

    public func loadConfig(_ id: UUID) throws -> String {
        try String(contentsOf: configURL(for: id), encoding: .utf8)
    }

    public func delete(_ id: UUID) throws {
        try fm.removeItem(at: directory(for: id))
    }

    private func writePrivate(_ data: Data, to url: URL) throws {
        try data.write(to: url, options: .atomic)
        try fm.setAttributes([.posixPermissions: 0o600], ofItemAtPath: url.path)
    }
}
```

```swift
import Foundation
import Security

public enum SecretField: String, CaseIterable, Sendable {
    case username, password, psk
}

public protocol SecretStore: AnyObject {
    func get(_ id: UUID, field: SecretField) -> String?
    /// Üres érték törli a mezőt.
    func set(_ value: String, for id: UUID, field: SecretField)
    func deleteAll(for id: UUID)
}

public extension SecretStore {
    func secrets(for id: UUID) -> Secrets {
        Secrets(username: get(id, field: .username), password: get(id, field: .password), psk: get(id, field: .psk))
    }
}

public final class KeychainSecretStore: SecretStore {
    let service: String

    public init(service: String = "macvpn") {
        self.service = service
    }

    private func query(_ id: UUID, _ field: SecretField) -> [String: Any] {
        [kSecClass as String: kSecClassGenericPassword,
         kSecAttrService as String: service,
         kSecAttrAccount as String: "\(id.uuidString).\(field.rawValue)"]
    }

    public func get(_ id: UUID, field: SecretField) -> String? {
        var q = query(id, field)
        q[kSecReturnData as String] = true
        q[kSecMatchLimit as String] = kSecMatchLimitOne
        var item: CFTypeRef?
        guard SecItemCopyMatching(q as CFDictionary, &item) == errSecSuccess, let data = item as? Data else { return nil }
        return String(data: data, encoding: .utf8)
    }

    public func set(_ value: String, for id: UUID, field: SecretField) {
        let q = query(id, field)
        SecItemDelete(q as CFDictionary)
        guard !value.isEmpty else { return }
        var add = q
        add[kSecValueData as String] = Data(value.utf8)
        add[kSecAttrAccessible as String] = kSecAttrAccessibleAfterFirstUnlock
        SecItemAdd(add as CFDictionary, nil)
    }

    public func deleteAll(for id: UUID) {
        for field in SecretField.allCases { SecItemDelete(query(id, field) as CFDictionary) }
    }
}

public final class InMemorySecretStore: SecretStore {
    private var values: [String: String] = [:]

    public init() {}

    public func get(_ id: UUID, field: SecretField) -> String? { values["\(id).\(field)"] }

    public func set(_ value: String, for id: UUID, field: SecretField) {
        values["\(id).\(field)"] = value.isEmpty ? nil : value
    }

    public func deleteAll(for id: UUID) {
        for field in SecretField.allCases { values["\(id).\(field)"] = nil }
    }
}
```

```swift
import Foundation

public struct CommandResult: Equatable, Sendable {
    public var exitCode: Int32
    public var stdout: String
    public var stderr: String

    public init(exitCode: Int32 = 0, stdout: String = "", stderr: String = "") {
        self.exitCode = exitCode
        self.stdout = stdout
        self.stderr = stderr
    }
}

public protocol CommandRunner: Sendable {
    func run(_ executable: String, _ arguments: [String], stdin: String?) async throws -> CommandResult
}

public struct ProcessCommandRunner: CommandRunner {
    public init() {}

    public func run(_ executable: String, _ arguments: [String], stdin: String?) async throws -> CommandResult {
        try await Task.detached(priority: .userInitiated) {
            let process = Process()
            process.executableURL = URL(fileURLWithPath: executable)
            process.arguments = arguments
            let out = Pipe(), err = Pipe(), inp = Pipe()
            process.standardOutput = out
            process.standardError = err
            process.standardInput = inp
            try process.run()
            if let stdin { inp.fileHandleForWriting.write(Data(stdin.utf8)) }
            try? inp.fileHandleForWriting.close()

            // stderr külön szálon, hogy egyik pipe se teljen be.
            let errBox = DataBox()
            let group = DispatchGroup()
            group.enter()
            DispatchQueue.global().async {
                errBox.data = err.fileHandleForReading.readDataToEndOfFile()
                group.leave()
            }
            let outData = out.fileHandleForReading.readDataToEndOfFile()
            group.wait()
            process.waitUntilExit()
            return CommandResult(exitCode: process.terminationStatus,
                                 stdout: String(decoding: outData, as: UTF8.self),
                                 stderr: String(decoding: errBox.data, as: UTF8.self))
        }.value
    }
}

private final class DataBox: @unchecked Sendable {
    var data = Data()
}
```

```swift
/// A kulcs a teljes parancssor szóközzel összefűzve, pl. "/usr/bin/sudo -n /usr/local/libexec/macvpn/mvpn status".
final class MockCommandRunner: CommandRunner, @unchecked Sendable {

        let arguments: [String]
        let stdin: String?
        var commandLine: String { ([executable] + arguments).joined(separator: " ") }
    }

    private let lock = NSLock()
    private var _calls: [Call] = []
    var responses: [String: CommandResult] = [:]

    var calls: [Call] { lock.lock(); defer { lock.unlock() }; return _calls }

    func run(_ executable: String, _ arguments: [String], stdin: String?) async throws -> CommandResult {
        let call = Call(executable: executable, arguments: arguments, stdin: stdin)
        lock.lock()
        _calls.append(call)
        let response = responses[call.commandLine]
        lock.unlock()
        return response ?? CommandResult()
    }
```

```swift
public struct SystemVPNService: Equatable, Identifiable, Sendable {
    public let id: String
    public let name: String
    public let typeLabel: String
    public let status: String

    public var connectionState: ConnectionState {
        switch status {
        case "Connected": return .connected(ip: nil, since: nil)
        case "Connecting": return .connecting
        case "Disconnecting": return .disconnecting
        default: return .disconnected
        }
    }
}

public enum ScutilParser {
    private static let regex = try! NSRegularExpression(
        pattern: #"^\*?\s*\(([^)]+)\)\s+([0-9A-Fa-f-]{36})\s+.*?"([^"]+)"\s+\[([^\]]+)\]"#)

    public static func parseList(_ output: String) -> [SystemVPNService] {
        output.components(separatedBy: "\n").compactMap { line in
            let ns = line as NSString
            guard let m = regex.firstMatch(in: line, range: NSRange(location: 0, length: ns.length)) else { return nil }
            let type = ns.substring(with: m.range(at: 4))
            guard let label = typeLabel(type) else { return nil }
            return SystemVPNService(id: ns.substring(with: m.range(at: 2)),
                                    name: ns.substring(with: m.range(at: 3)),
                                    typeLabel: label,
                                    status: ns.substring(with: m.range(at: 1)))
        }
    }

    /// nil: nem VPN (pl. modem).
    static func typeLabel(_ raw: String) -> String? {
        switch raw {
        case "PPP:L2TP": return "L2TP"
        case "IPSec": return "Cisco IPsec"
        case "IKEv2": return "IKEv2"
        case "VPN:com.wireguard.macos": return "WireGuard (app)"
        case "VPN:net.tunnelblick.tunnelblick": return "OpenVPN (app)"
        default: return raw.hasPrefix("VPN:") ? "VPN" : nil
        }
    }
```

```swift
public enum LogFilter: String, CaseIterable, Identifiable, Sendable {
    case all = "Minden"
    case commands = "Parancsok"
    case errors = "Hibák"
    public var id: String { rawValue }

public struct LogLine: Equatable, Sendable {
    public enum Kind: Sendable { case header, step, command, error, output }

    public let text: String
    public let kind: Kind

    static let errorMarkers = ["AUTH_FAILED", "TLS handshake failed", "TLS Error", "Exiting due to fatal error",
                               "establishing connection failed", "failed to establish", "Connection refused"]

    public init(_ text: String) {
        self.text = text
        if text.hasPrefix("═══") { kind = .header }
        else if text.hasPrefix("▶ [mvpn]") { kind = .step }
        else if text.hasPrefix("$ ") || text.hasPrefix("[#] ") { kind = .command }
        else if text.hasPrefix("✗") || Self.errorMarkers.contains(where: { text.contains($0) }) { kind = .error }
        else { kind = .output }
    }

    public func matches(_ filter: LogFilter) -> Bool {
        switch filter {
        case .all: return true
        case .commands: return kind == .command || kind == .step || kind == .header
        case .errors: return kind == .error
        }
    }
}
```

```swift
import Foundation

/// Offsetből olvassa a fájl új, teljes sorait. Csonkoláskor elölről kezdi.
public final class LogTailer {
    public let url: URL
    private var offset: UInt64 = 0
    private var partial = Data()

    public init(url: URL) {
        self.url = url
    }

    public func readNew() -> [String] {
        guard let handle = try? FileHandle(forReadingFrom: url) else { return [] }
        defer { try? handle.close() }
        let size = (try? handle.seekToEnd()) ?? 0
        if size < offset {
            offset = 0
            partial = Data()
        }
        guard size > offset else { return [] }
        try? handle.seek(toOffset: offset)
        let data = (try? handle.readToEnd()) ?? Data()
        offset += UInt64(data.count)
        partial.append(data)
        guard let lastNewline = partial.lastIndex(of: UInt8(ascii: "\n")) else { return [] }
        let complete = partial[partial.startIndex..<lastNewline]
        partial = Data(partial[(lastNewline + 1)...])
        return String(decoding: complete, as: UTF8.self).components(separatedBy: "\n")
    }
}
```

```swift
    let readFile: (String) -> String?

                readFile: @escaping (String) -> String? = { try? String(contentsOfFile: $0, encoding: .utf8) },

        self.fileExists = fileExists
        self.readFile = readFile
        self.bundledRootDir = bundledRootDir
    }

    public static func candidates(tool: String, prefix: String) -> [String] {
        if tool == "charon" {
            return ["\(prefix)/opt/strongswan/libexec/ipsec/charon", "\(prefix)/opt/strongswan/libexec/charon"]
        }
        return ["\(prefix)/bin/\(tool)", "\(prefix)/sbin/\(tool)"]
    }

        let prefix = Self.brewPrefixes.first { fileExists("\($0)/bin/brew") }
        var missing: [String] = []
        for (_, tools) in Self.requiredTools {
            for tool in tools where !missing.contains(tool) {
                let found = prefix.map { p in Self.candidates(tool: tool, prefix: p).contains(where: fileExists) } ?? false
                if !found { missing.append(tool) }
            }
        }
        let trim = { (s: String?) in s?.trimmingCharacters(in: .whitespacesAndNewlines) }
        let installed = trim(readFile(MvpnClient.installDir + "/VERSION"))
        let bundled = trim(bundledRootDir.flatMap { readFile($0.appendingPathComponent("VERSION").path) })
        let sudo = (try? await runner.run("/usr/bin/sudo", ["-n", MvpnClient.executable, "version"], stdin: nil))?.exitCode == 0
        return SetupReport(brewPrefix: prefix, missingTools: missing, installedRootVersion: installed,
                           bundledRootVersion: bundled, sudoWorks: sudo)
    }
```

```swift
import Foundation
import Combine

public enum ConnectionEvent: Equatable, Sendable {
    case failed(connectionID: String, name: String, reason: String)
    case dropped(connectionID: String, name: String)
}

@MainActor
public final class ConnectionManager: ObservableObject {
    @Published public private(set) var connections: [Connection] = []
    /// Csak azok a rendszer VPN-ek, amelyeket nem az app kezel.
    @Published public private(set) var systemServices: [SystemVPNService] = []
    @Published public private(set) var states: [String: ConnectionState] = [:]

    public var onEvent: ((ConnectionEvent) -> Void)?

    public let store: ConnectionStore
    public let secrets: SecretStore
    let mvpn: MvpnClient
    let native: NativeVPNController
    private var pending: Set<String> = []

    public init(store: ConnectionStore, secrets: SecretStore, mvpn: MvpnClient, native: NativeVPNController) {
        self.store = store
        self.secrets = secrets
        self.mvpn = mvpn
        self.native = native
    }

    public func reload() {
        connections = (try? store.list()) ?? []
    }

    public func state(for key: String) -> ConnectionState { states[key] ?? .disconnected }

    public var hasActive: Bool { states.values.contains { $0.isActive } }

    public var hasFailure: Bool {
        states.values.contains { if case .failed = $0 { return true } else { return false } }
    }

    public func refresh() async {
        let root = (try? await mvpn.status()) ?? []
        let services = (try? await native.list()) ?? []
        var next: [String: ConnectionState] = [:]
        var managedNativeNames = Set<String>()

        for c in connections {
            let key = c.id.uuidString
            let previous = states[key]
            var current: ConnectionState
            if c.kind.isNative {
                let name = c.nativeServiceName ?? c.name
                managedNativeNames.insert(name)
                current = services.first { $0.name == name }?.connectionState ?? .disconnected
            } else if let status = root.first(where: { $0.id == c.id }) {
                current = status.connectionState
            } else {
                current = .disconnected
            }
            if pending.contains(key) {
                current = previous ?? .connecting
            } else if case .failed? = previous, current == .disconnected {
                current = previous!
            }
            if case .connected? = previous, case .failed = current {
                onEvent?(.dropped(connectionID: key, name: c.name))
            }
            next[key] = current
        }

        let unmanaged = services.filter { !managedNativeNames.contains($0.name) }
        for s in unmanaged { next[s.id] = s.connectionState }
        systemServices = unmanaged
        states = next
    }

    public func connect(_ c: Connection, debug: Bool) async {
        let key = c.id.uuidString
        pending.insert(key)
        states[key] = .connecting
        do {
            if c.kind.isNative {
                let name = c.nativeServiceName ?? c.name
                guard let service = try await native.list().first(where: { $0.name == name }) else {
                    throw MvpnError.failed("A profil nincs telepítve: \(name)")
                }
                try await native.start(serviceID: service.id)
            } else {
                try await mvpn.up(kind: c.kind, id: c.id, secrets: secrets.secrets(for: c.id), debug: debug)
            }
            pending.remove(key)
            states[key] = .disconnected
            await refresh()
        } catch {
            pending.remove(key)
            let reason = error.localizedDescription
            states[key] = .failed(reason)
            onEvent?(.failed(connectionID: key, name: c.name, reason: reason))
        }
    }

    public func disconnect(_ c: Connection) async {
        let key = c.id.uuidString
        states[key] = .disconnecting
        do {
            if c.kind.isNative {
                let name = c.nativeServiceName ?? c.name
                if let service = try await native.list().first(where: { $0.name == name }) {
                    try await native.stop(serviceID: service.id)
                }
            } else {
                try await mvpn.down(kind: c.kind, id: c.id)
            }
        } catch {
            onEvent?(.failed(connectionID: key, name: c.name, reason: error.localizedDescription))
        }
        states[key] = .disconnected
        await refresh()
    }

    public func connectSystem(_ s: SystemVPNService) async {
        states[s.id] = .connecting
        do { try await native.start(serviceID: s.id) } catch {
            states[s.id] = .failed(error.localizedDescription)
            onEvent?(.failed(connectionID: s.id, name: s.name, reason: error.localizedDescription))
            return
        }
        await refresh()
    }

    public func disconnectSystem(_ s: SystemVPNService) async {
        states[s.id] = .disconnecting
        try? await native.stop(serviceID: s.id)
        await refresh()
    }

    /// Aktív managed kapcsolatok bontása (kilépéskor).
    public func disconnectAll() async {
        for c in connections where state(for: c.id.uuidString).isActive {
            await disconnect(c)
        }
    }
}
```

```bash
# Literális (nem regex) csere: MVPN_MASK_1..3 → ***
mask() {
  awk 'BEGIN { n = 0; for (i = 1; i <= 3; i++) { v = ENVIRON["MVPN_MASK_" i]; if (length(v) >= 3) s[++n] = v } }
       { line = $0
         for (i = 1; i <= n; i++) {
           out = ""
           while ((p = index(line, s[i])) > 0) { out = out substr(line, 1, p - 1) "***"; line = substr(line, p + length(s[i])) }
           line = out line
         }
         print line; fflush() }'
}

}
```

```bash
#!/bin/bash
# OpenVPN up/down hook (root). Hívás: ovpn-dns <run-dir> <log-fájl>, a típus a $script_type env-ből.
# A szerver által küldött DNS-t minden hálózati szolgáltatásra beállítja (mint a wg-quick), down-nál visszaállítja.
# A szerver által küldött értékeket nem eval-oljuk: printenv + szigorú formátum-ellenőrzés.
export PATH="/usr/bin:/bin:/usr/sbin:/sbin"
RUN="$1"
LOG="$2"
[ -d "$RUN" ] || exit 0
log() { printf '%s\n' "$*" >>"$LOG"; }

case "$script_type" in
  up)
    printf '%s\n' "$dev" >"$RUN/iface"
    printf '%s\n' "$ifconfig_local" >"$RUN/ip"
    DNS=(); DOMS=()
    i=1
    while o=$(printenv "foreign_option_$i"); do
      case "$o" in
        "dhcp-option DNS "*)    v="${o#dhcp-option DNS }";    [[ "$v" =~ ^[0-9A-Fa-f:.]+$ ]] && DNS+=("$v") ;;
        "dhcp-option DOMAIN "*) v="${o#dhcp-option DOMAIN }"; [[ "$v" =~ ^[A-Za-z0-9.-]+$ ]] && DOMS+=("$v") ;;
      esac
      i=$((i + 1))
    done
    [ ${#DNS[@]} -eq 0 ] && { log "▶ [mvpn] a szerver nem küldött DNS-t"; exit 0; }
    : >"$RUN/dns-backup"
    networksetup -listallnetworkservices | tail -n +2 | sed 's/^\*//' | while IFS= read -r svc; do
      cur=$(networksetup -getdnsservers "$svc" | tr '\n' ' ')
      case "$cur" in *"aren't any"*) cur="Empty" ;; esac
      curs=$(networksetup -getsearchdomains "$svc" | tr '\n' ' ')
      case "$curs" in *"aren't any"*) curs="Empty" ;; esac
      printf '%s\t%s\t%s\n' "$svc" "$cur" "$curs" >>"$RUN/dns-backup"
      log "\$ networksetup -setdnsservers \"$svc\" ${DNS[*]}"
      networksetup -setdnsservers "$svc" "${DNS[@]}"
      if [ ${#DOMS[@]} -gt 0 ]; then
        log "\$ networksetup -setsearchdomains \"$svc\" ${DOMS[*]}"
        networksetup -setsearchdomains "$svc" "${DOMS[@]}"
      fi
    done
    ;;
  down)
    [ -f "$RUN/dns-backup" ] || exit 0
    while IFS=$'\t' read -r svc dns doms; do
      log "\$ networksetup -setdnsservers \"$svc\" $dns"
      networksetup -setdnsservers "$svc" $dns
      networksetup -setsearchdomains "$svc" $doms
    done <"$RUN/dns-backup"
    rm -f "$RUN/dns-backup"
    ;;
esac
exit 0
```

```bash
status_one() { # $1 = run dir → "state iface ip"
  local d="$1" kind state="failed" iface="" ip="" pid conn real
  kind=$(cat "$d/kind")
  case "$kind" in
    openvpn)
      pid=$(cat "$d/pid" 2>/dev/null)
      if [ -n "$pid" ] && kill -0 "$pid" 2>/dev/null; then
        if [ -f "$d/iface" ]; then state="connected"; else state="connecting"; fi
      fi
      iface=$(cat "$d/iface" 2>/dev/null); ip=$(cat "$d/ip" 2>/dev/null)
      ;;
    wireguard)
      iface=$(cat "$d/iface" 2>/dev/null)
      real=$(cat "/var/run/wireguard/$iface.name" 2>/dev/null)
      if [ -n "$real" ] && ifconfig "$real" >/dev/null 2>&1; then
        state="connected"
        ip=$(ifconfig "$real" | awk '/inet /{ print $2; exit }')
        iface="$real"
      fi
      ;;
    strongswan)
      conn=$(cat "$d/conn" 2>/dev/null)
      local sw; sw=$(find_tool swanctl)
      if [ -n "$sw" ] && [[ "$conn" =~ ^[A-Za-z0-9_.-]+$ ]]; then
        local sas; sas=$("$sw" --list-sas --ike "$conn" 2>/dev/null)
        case "$sas" in
          *ESTABLISHED*) state="connected" ;;
          *CONNECTING*)  state="connecting" ;;
        esac
      fi
      ;;
  esac
  [[ "$iface" =~ ^[A-Za-z0-9]*$ ]] || iface=""
  [[ "$ip" =~ ^[0-9A-Fa-f:.]*$ ]] || ip=""
  printf '%s %s %s\n' "$state" "${iface:--}" "${ip:--}"
}
```

```swift
import Foundation

public enum RootInstaller {
    public struct InstallError: LocalizedError {
        public let message: String
        public var errorDescription: String? { message }
    }

    static func shellQuote(_ s: String) -> String { "'" + s.replacingOccurrences(of: "'", with: "'\\''") + "'" }

    public static func installScript(sourceDir: URL, user: String) throws -> String {
        guard user.range(of: "^[A-Za-z_][A-Za-z0-9_.-]*$", options: .regularExpression) != nil else {
            throw InstallError(message: "Érvénytelen felhasználónév: \(user)")
        }
        let src = { (name: String) in shellQuote(sourceDir.appendingPathComponent(name).path) }
        let dir = MvpnClient.installDir
        return """
        #!/bin/bash
        set -euo pipefail
        install -d -o root -g wheel -m 755 \(dir)
        install -o root -g wheel -m 755 \(src("mvpn")) \(dir)/mvpn
        install -o root -g wheel -m 755 \(src("ovpn-dns")) \(dir)/ovpn-dns
        install -o root -g wheel -m 755 \(src("mvpn-validate")) \(dir)/mvpn-validate
        install -o root -g wheel -m 644 \(src("VERSION")) \(dir)/VERSION
        install -d -o root -g wheel -m 755 \(MvpnClient.logDir)
        install -d -o root -g wheel -m 700 /var/run/macvpn
        TMP=$(mktemp)
        printf '%s\\n' '\(user) ALL=(root) NOPASSWD: \(dir)/mvpn' > "$TMP"
        visudo -c -f "$TMP"
        install -o root -g wheel -m 440 "$TMP" /etc/sudoers.d/macvpn
        rm -f "$TMP"

        """
    }

    public static func appleScript(scriptPath: String) -> String {
        let escaped = scriptPath.replacingOccurrences(of: "\\", with: "\\\\").replacingOccurrences(of: "\"", with: "\\\"")
        return "do shell script \"/bin/bash \" & quoted form of \"\(escaped)\" with administrator privileges"
    }

    /// Egy admin jelszó-bekéréssel telepíti a root scripteket és a sudoers szabályt.
    public static func install(sourceDir: URL, user: String = NSUserName(), runner: CommandRunner) async throws {
        let script = try installScript(sourceDir: sourceDir, user: user)
        let url = FileManager.default.temporaryDirectory.appendingPathComponent("macvpn-install-\(UUID().uuidString).sh")
        try script.write(to: url, atomically: true, encoding: .utf8)
        defer { try? FileManager.default.removeItem(at: url) }
        let result = try await runner.run("/usr/bin/osascript", ["-e", appleScript(scriptPath: url.path)], stdin: nil)
        guard result.exitCode == 0 else {
            let msg = result.stderr.trimmingCharacters(in: .whitespacesAndNewlines)
            throw InstallError(message: msg.contains("-128") ? "A telepítés megszakítva" : "Telepítési hiba: \(msg)")
        }
    }
}
```

```swift
/// Natív (L2TP/Cisco) debug csatlakozáskor a rendszernaplót gyűjti, legfeljebb 60 mp-ig.
@MainActor
final class NativeLogCollector: ObservableObject {
    @Published private(set) var lines: [String] = []
    private var process: Process?

    static let predicate = #"process IN {"racoon","pppd","neagent","nesessionmanager"} OR subsystem == "com.apple.networkextension""#

    func start() {
        stop()
        lines = ["═══ \(Date().formatted(date: .numeric, time: .standard)) natív debug ═══",
                 "$ log stream --style compact --predicate '\(Self.predicate)'"]
        let p = Process()
        p.executableURL = URL(fileURLWithPath: "/usr/bin/log")
        p.arguments = ["stream", "--style", "compact", "--predicate", Self.predicate]
        let pipe = Pipe()
        p.standardOutput = pipe
        p.standardError = pipe
        pipe.fileHandleForReading.readabilityHandler = { [weak self] handle in
            let data = handle.availableData
            guard !data.isEmpty else { return }
            let text = String(decoding: data, as: UTF8.self)
            Task { @MainActor in
                self?.lines.append(contentsOf: text.split(separator: "\n").map(String.init))
            }
        }
        do {
            try p.run()
            process = p
            Task { @MainActor [weak self] in
                try? await Task.sleep(for: .seconds(60))
                self?.stop()
            }
        } catch {
            lines.append("✗ log stream nem indult: \(error.localizedDescription)")
        }
    }

    func stop() {
        guard let p = process else { return }
        (p.standardOutput as? Pipe)?.fileHandleForReading.readabilityHandler = nil
        if p.isRunning { p.terminate() }
        process = nil
        lines.append("▶ [mvpn] naplógyűjtés leállítva")
    }
}
```

```swift
/// Export a config eredeti formátumában. Jelszó, PSK, felhasználónév nem kerül bele.
public enum ConfigExporter {
    public static func fileName(for c: Connection) -> String {
        let safe = c.name.replacingOccurrences(of: "/", with: "-").replacingOccurrences(of: ":", with: "-")
        switch c.kind {
        case .openvpn: return "\(safe).ovpn"
        case .wireguard: return "\(safe).conf"
        case .strongswan: return "\(safe).swanctl.conf"
        case .l2tp, .cisco: return "\(safe).mobileconfig"
        }
    }

    public static func export(_ c: Connection, store: ConnectionStore) throws -> Data {
        if c.kind.isNative {
            return try MobileconfigGenerator.generate(connection: c, password: "", psk: "")
        }
        return Data(try store.loadConfig(c.id).utf8)
    }

    public static func warning(for kind: VPNKind) -> String? {
        kind == .wireguard
            ? "A WireGuard config tartalmazza a privát kulcsot (PrivateKey). Csak megbízható helyre mentsd vagy add át."
            : nil
    }
}
```

```swift
import SwiftUI
import AppKit

@MainActor
final class LogViewModel: ObservableObject {
    @Published private(set) var lines: [LogLine] = []
    private var tailer: LogTailer?
    private var timer: Timer?
    private var cancellable: AnyCancellable?
    static let maxLines = 5000

    func start(connection: Connection, collector: NativeLogCollector?) {
        stop()
        if connection.kind.isNative {
            guard let collector else {
                lines = [LogLine("▶ [mvpn] Natív kapcsolatnál csak debug csatlakozáskor van napló.")]
                return
            }
            cancellable = collector.$lines.sink { [weak self] raw in
                self?.lines = raw.suffix(Self.maxLines).map(LogLine.init)
            }
            return
        }
        let tailer = LogTailer(url: MvpnClient.logURL(for: connection.id))
        self.tailer = tailer
        append(tailer.readNew())
        timer = Timer.scheduledTimer(withTimeInterval: 0.5, repeats: true) { [weak self] _ in
            Task { @MainActor in self?.append(self?.tailer?.readNew() ?? []) }
        }
    }

    private func append(_ new: [String]) {
        guard !new.isEmpty else { return }
        lines.append(contentsOf: new.map(LogLine.init))
        if lines.count > Self.maxLines { lines.removeFirst(lines.count - Self.maxLines) }
    }

    func stop() {
        timer?.invalidate()
        timer = nil
        tailer = nil
        cancellable = nil
    }
}
```

```swift
import Foundation
import MacVPNCore

/// `brew install …` futtatása user jogon, élő kimenettel.
@MainActor
final class BrewInstaller: ObservableObject {
    @Published private(set) var output = ""
    @Published private(set) var running = false
    @Published private(set) var finishedCount = 0
    @Published private(set) var lastExitCode: Int32?

    func install(brewPrefix: String) {
        guard !running else { return }
        let p = Process()
        p.executableURL = URL(fileURLWithPath: brewPrefix + "/bin/brew")
        p.arguments = ["install"] + SetupChecker.brewPackages
        var env = ProcessInfo.processInfo.environment
        env["PATH"] = "\(brewPrefix)/bin:\(brewPrefix)/sbin:/usr/bin:/bin:/usr/sbin:/sbin"
        env["HOMEBREW_NO_ENV_HINTS"] = "1"
        p.environment = env
        let pipe = Pipe()
        p.standardOutput = pipe
        p.standardError = pipe
        pipe.fileHandleForReading.readabilityHandler = { [weak self] handle in
            let data = handle.availableData
            guard !data.isEmpty else { return }
            let text = String(decoding: data, as: UTF8.self)
            Task { @MainActor in self?.output += text }
        }
        p.terminationHandler = { [weak self] proc in
            Task { @MainActor in
                pipe.fileHandleForReading.readabilityHandler = nil
                self?.running = false
                self?.lastExitCode = proc.terminationStatus
                self?.output += proc.terminationStatus == 0 ? "\n✓ Kész.\n" : "\n✗ brew hiba (\(proc.terminationStatus))\n"
                self?.finishedCount += 1
            }
        }
        output = "$ brew install \(SetupChecker.brewPackages.joined(separator: " "))\n"
        running = true
        do { try p.run() } catch {
            running = false
            output += "✗ \(error.localizedDescription)\n"
        }
    }
}
```

```bash
# MacVPN.app összeállítása és DMG csomagolás (ad-hoc aláírás, Developer ID nélkül).

VERSION=$(/usr/libexec/PlistBuddy -c "Print CFBundleShortVersionString" Resources/Info.plist)

cp "$BIN" "$APP/Contents/MacOS/MacVPN"
cp Resources/Info.plist "$APP/Contents/Info.plist"
cp Resources/root/mvpn Resources/root/ovpn-dns Resources/root/VERSION "$APP/Contents/Resources/root/"
cp "$VALIDATE" "$APP/Contents/Resources/root/mvpn-validate"
chmod 755 "$APP/Contents/Resources/root/mvpn" "$APP/Contents/Resources/root/ovpn-dns" "$APP/Contents/Resources/root/mvpn-validate"
if [ -f Resources/AppIcon.icns ]; then
  cp Resources/AppIcon.icns "$APP/Contents/Resources/"
  /usr/libexec/PlistBuddy -c "Add :CFBundleIconFile string AppIcon" "$APP/Contents/Info.plist"
fi

echo "▶ ad-hoc aláírás"
codesign --force --deep --sign - "$APP"
codesign --verify --verbose "$APP"

echo "▶ DMG"
STAGE="dist/dmg"
mkdir -p "$STAGE"
cp -R "$APP" "$STAGE/"
ln -s /Applications "$STAGE/Applications"
hdiutil create -volname "MacVPN" -srcfolder "$STAGE" -ov -format UDZO "dist/MacVPN-$VERSION.dmg" >/dev/null
```
